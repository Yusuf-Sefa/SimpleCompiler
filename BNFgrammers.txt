Program    -> StatementList
StatementList -> Statement StatementList | ε
Statement  -> VariableDecl | Assignment | PrintStmt | IfStmt | WhileStmt

VariableDecl -> KEYWORD IDENTIFIER ";"
Assignment   -> IDENTIFIER "=" Expression ";"
PrintStmt    -> "print" "(" ( MathExpression | STRING_LITERAL ) ")" ";"

IfStmt       -> "if" "(" Expression ")" "{" StatementList "}" ElseStmt
ElseStmt     -> "else" "{" "StatementList" "}" | ε
WhileStmt    -> "while" "(" Expression ")" "{" StatementList "}"

Expression   -> MathExpression RelationalOp MathExpression | MathExpression
MathExpression -> Term (("+" | "-") Term)*
RelationalOp   -> ">" | "<" | "==" | "!=" | ">=" | "<="
Term         -> Factor (("*" | "/") Factor)*
Factor       -> IDENTIFIER | INT_LITERAL | FLOAT_LITERAL | "(" Expression ")"
